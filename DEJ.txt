              function formatBytes(bytes, decimals = 1) {
                  if (!bytes || bytes === 0) return '0 KB';
                  const k = 1024;

              function reassignElementToPage(elem, itemType) {
                  const elemRect = elem.getBoundingClientRect();
                  const centerPointY = elemRect.top + elemRect.height / 2;

                  let targetPageIndex = 0;
                  let minDistance = Infinity;

                  pages.forEach((p, idx) => {
                      const pRect = p.card.getBoundingClientRect();
                      const pCenterY = pRect.top + pRect.height / 2;
                      const dist = Math.abs(centerPointY - pCenterY);
                      if (dist < minDistance) {
                          minDistance = dist;
                          targetPageIndex = idx;
                      }
                  });

                  const targetPage = pages[targetPageIndex];
                  const targetCard = targetPage.card;

                  let currentOwnerIndex = pages.findIndex(p => p.card === elem.parentElement);

                  if (currentOwnerIndex !== -1 && currentOwnerIndex !== targetPageIndex) {
                      const oldPage = pages[currentOwnerIndex];

                      if (itemType === 'image') {
                          const entryIdx = oldPage.imageBoxes.findIndex(ib => ib.elem === elem);
                          if (entryIdx !== -1) {
                              const entry = oldPage.imageBoxes.splice(entryIdx, 1)[0];
                              targetPage.imageBoxes.push(entry);
                          }
                      } else if (itemType === 'text') {
                          const entryIdx = oldPage.textBoxes.findIndex(tb => tb.elem === elem);
                          if (entryIdx !== -1) {
                              const entry = oldPage.textBoxes.splice(entryIdx, 1)[0];
                              targetPage.textBoxes.push(entry);
                          }
                      }

                      targetCard.appendChild(elem);

                      const tRect = targetCard.getBoundingClientRect();
                      const relLeft = (elemRect.left - tRect.left) / scale;
                      const relTop = (elemRect.top - tRect.top) / scale;

                      elem.style.left = relLeft + 'px';

                  }

              function getFolderPath(folderId) {
                  let path = [{ id: 'root', name: '🏠 Home' }];
                  if (folderId === 'root') return path;

                  let stack = [];
                  let curr = appState.folders.find(f => f.id === folderId);
                  while (curr) {
                      stack.unshift(curr);
                      curr = appState.folders.find(f => f.id === curr.parentId);
                  }
                  return path.concat(stack);
              }

              function deleteFolderRecursive(folderId) {
                  const subFolders = appState.folders.filter(f => f.parentId === folderId);
                  subFolders.forEach(sf => deleteFolderRecursive(sf.id));

                  appState.notes = appState.notes.filter(n => n.folderId !== folderId);
                  appState.folders = appState.folders.filter(f => f.id !== folderId);
              }

              function openNoteStudio(noteId) {
                  currentNoteId = noteId;
                  const note = appState.notes.find(n => n.id === noteId);
                  if (!note) return;

                  currentDocumentTitle = note.title;
                  document.getElementById('docTitleBadge').innerText = note.title;

                  if (note.pageSettings) {
                      pageSettings = Object.assign({ bgColor: '#ffffff', showPageNumbers: true, bgImageDataUrl: null }, note.pageSettings);
                      document.getElementById('pageSizeSelect').value = pageSettings.size;
                      document.getElementById('pageOrientationSelect').value = pageSettings.orientation;
                      document.getElementById('borderStyleSelect').value = pageSettings.borderStyle;
                      document.getElementById('borderColorInput').value = pageSettings.borderColor;
                      document.getElementById('borderMarginInput').value = pageSettings.borderMargin;
                      document.getElementById('pageMarginGuideSelect').value = pageSettings.pageMarginGuide;
                      document.getElementById('ruledLineCountInput').value = pageSettings.ruledLineCount;
                      document.getElementById('bgSelect').value = pageSettings.background;
                      document.getElementById('pageBgColorInput').value = pageSettings.bgColor || '#ffffff';
                      document.getElementById('pageBgPresetSelect').value = pageSettings.bgColor || '#ffffff';
                      document.getElementById('showPageNumbersSelect').value = (pageSettings.showPageNumbers !== false).toString();
                  }

                  pagesContainer.innerHTML = '';
                  pages = [];

                  const count = note.pageCount || 1;
                  for (let i = 0; i < count; i++) {
                      createNewPage(true);
                      const pData = note.pagesData ? note.pagesData[i] : null;
                      if (pData) {
                          if (pData.inkData) {
                              const img = new Image();
                              const pageObj = pages[i];
                              img.src = pData.inkData;
                              img.onload = () => {
                                  pageObj.baseLayerCanvas.width = pageObj.canvas.width;
                                  pageObj.baseLayerCanvas.height = pageObj.canvas.height;
                                  pageObj.baseLayerCtx.imageSmoothingEnabled = false;
                                  pageObj.baseLayerCtx.drawImage(img, 0, 0, pageObj.canvas.width, pageObj.canvas.height);

                                  pageObj.ctx.save();
                                  pageObj.ctx.setTransform(1, 0, 0, 1, 0, 0);
                                  pageObj.ctx.imageSmoothingEnabled = false;
                                  pageObj.ctx.clearRect(0, 0, pageObj.canvas.width, pageObj.canvas.height);
                                  pageObj.ctx.drawImage(pageObj.baseLayerCanvas, 0, 0, pageObj.canvas.width, pageObj.canvas.height);
                                  pageObj.ctx.restore();
                              };
                          }
                          if (pData.imageBoxes) {
                              pData.imageBoxes.forEach(sImg => {
                                  const img = new Image();
                                  img.onload = () => insertImageToPage(i, img, sImg.src, sImg);
                                  img.src = sImg.src;
                              });
                          }
                          if (pData.textBoxes) {
                              pData.textBoxes.forEach(sTb => createTextBox(i, sTb.left, sTb.top, sTb.content, sTb.color, sTb.width));
                          }
                      }
                  }

                  dashboardView.style.display = 'none';
                  textStudioView.style.display = 'none';
                  studioView.style.display = 'block';
                  centerPages();
                  updateDocSizeBadge();

                  startAutoSaveTimer();
              }

              function updateDocSizeBadge() {

              async function executeNoteSave(isSilent = false) {
                  if (!currentNoteId) return;

                  const note = appState.notes.find(n => n.id === currentNoteId);
                  if (!note) return;

                  if (note.type === 'text') {
                      note.title = currentDocumentTitle;
                      note.headerTitle = document.getElementById('textNoteHeaderInput').value;
                      note.bannerImage = currentBannerImage || DEFAULT_BANNER_SVG;
                      note.content = document.getElementById('richEditor').innerHTML;
                      note.bgStyle = document.getElementById('textPaperBgStyleSelect').value;
                      note.patternType = document.getElementById('textProceduralPatternSelect').value;
                      note.paperColor = document.getElementById('textPaperColorPicker').value;

                      const textImgBoxes = textPaperCard.querySelectorAll('.text-img-box');
                      note.imagesData = Array.from(textImgBoxes).map(box => {
                          const img = box.querySelector('img');
                          const canvas = box.querySelector('canvas');
                          return {
                              left: parseFloat(box.style.left) || 40,
                              top: parseFloat(box.style.top) || 120,
                              width: box.offsetWidth,
                              height: box.offsetHeight,
                              src: img ? img.src : '',
                              inkData: canvas ? canvas.toDataURL() : ''
                          };
                      });

                      try {
                          document.body.classList.add('exporting-clean-mode');
                          const canvas = await html2canvas(textPaperCard, {
                              scale: 0.5,
                              useCORS: true,
                              backgroundColor: null
                          });
                          document.body.classList.remove('exporting-clean-mode');
                          note.thumbnail = canvas.toDataURL('image/jpeg', 0.6);
                      } catch(e) {
                          console.error("Text note preview generation error:", e);
                          document.body.classList.remove('exporting-clean-mode');
                      }

                  } else {
                      note.title = currentDocumentTitle;
                      note.pageSettings = Object.assign({}, pageSettings);
                      note.pageCount = pages.length;
                      note.pagesData = pages.map(p => ({
                          inkData: p.canvas.toDataURL(),
                          imageBoxes: p.imageBoxes.map(ib => ({
                              src: ib.imgElem.src,
                              left: parseFloat(ib.elem.style.left) || 60,
                              top: parseFloat(ib.elem.style.top) || 60,
                              width: ib.elem.offsetWidth,
                              height: ib.elem.offsetHeight,
                              imageInkData: ib.inkCanvas.toDataURL()
                          })),
                          textBoxes: p.textBoxes.map(tb => ({
                              left: parseFloat(tb.elem.style.left) || 0,
                              top: parseFloat(tb.elem.style.top) || 0,
                              width: parseFloat(tb.elem.style.width) || 220,
                              content: tb.contentElem.innerText,
                              color: tb.contentElem.style.color || currentColor
                          }))
                      }));

                      if (pages.length > 0) {
                          note.thumbnail = renderPageToCanvas(0).toDataURL('image/jpeg', 0.6);
                      }
                  }

                  await saveStorageTree();
                  updateDocSizeBadge();

                  if (!isSilent) {
                      showToast(`Saved to "${currentDocumentTitle}"`);
                  }
              }

                  const bgCanvas = document.createElement('canvas');
                  bgCanvas.className = 'bg-canvas';
                  card.appendChild(bgCanvas);

                  const inkCanvas = document.createElement('canvas');
                  inkCanvas.className = 'ink-canvas';
                  card.appendChild(inkCanvas);

                  pagesContainer.appendChild(card);

                  const baseLayerCanvas = document.createElement('canvas');

                  const pageObj = {
                      card: card,
                      pageTag: pageTag,
                      marginOverlay: marginOverlay,
                      bgCanvas: bgCanvas,
                      bgCtx: bgCanvas.getContext('2d'),
                      canvas: inkCanvas,
                      ctx: inkCanvas.getContext('2d', { willReadFrequently: true }),
                      baseLayerCanvas: baseLayerCanvas,
                      baseLayerCtx: baseLayerCanvas.getContext('2d'),
                      textBoxes: [],
                      imageBoxes: []
                  };

                  pages.push(pageObj);
                  applyLayoutToPage(pageIndex);
              }

              function getPageDimensions() {
                  const base = PAGE_SIZES[pageSettings.size];
                  return pageSettings.orientation === 'portrait'
                      ? { width: base.width, height: base.height }
                      : { width: base.height, height: base.width };
              }

              function getRuledLineHeight() {
                  const dims = getPageDimensions();
                  const lineCount = parseInt(pageSettings.ruledLineCount) || 25;
                  const totalSlots = lineCount + 4;
                  return dims.height / totalSlots;
              }

              function applyLayoutToPage(index) {
                  const page = pages[index];
                  const dims = getPageDimensions();

                  page.card.style.width = dims.width + 'px';
                  page.card.style.height = dims.height + 'px';

                  page.bgCanvas.width = dims.width * RENDER_SCALE;
                  page.bgCanvas.height = dims.height * RENDER_SCALE;
                  page.bgCanvas.style.width = dims.width + 'px';
                  page.bgCanvas.style.height = dims.height + 'px';
                  page.bgCtx.scale(RENDER_SCALE, RENDER_SCALE);

                  page.canvas.width = dims.width * RENDER_SCALE;
                  page.canvas.height = dims.height * RENDER_SCALE;
                  page.canvas.style.width = dims.width + 'px';
                  page.canvas.style.height = dims.height + 'px';
                  page.ctx.scale(RENDER_SCALE, RENDER_SCALE);

                  page.baseLayerCanvas.width = dims.width * RENDER_SCALE;
                  page.baseLayerCanvas.height = dims.height * RENDER_SCALE;

                  const offset = parseInt(pageSettings.borderMargin) || 0;
                  const color = pageSettings.borderColor;
                  page.card.style.padding = offset + 'px';

                  page.pageTag.style.display = pageSettings.showPageNumbers !== false ? 'block' : 'none';

                  switch(pageSettings.borderStyle) {
                      case 'thin': page.card.style.border = `2px solid ${color}`; break;
                      case 'thick': page.card.style.border = `5px solid ${color}`; break;
                      case 'double': page.card.style.border = `7px double ${color}`; break;
                      case 'dashed': page.card.style.border = `3px dashed ${color}`; break;
                      default: page.card.style.border = 'none'; break;
                  }

                  renderBackgroundPattern(page.bgCtx, dims.width, dims.height);
                  renderMarginGuideOverlay(page.marginOverlay, dims);

                  const ruleH = getRuledLineHeight();
                  page.textBoxes.forEach(tb => {
                      tb.contentElem.style.fontSize = (ruleH * 0.72) + 'px';
                      tb.contentElem.style.lineHeight = ruleH + 'px';
                      tb.contentElem.style.minHeight = ruleH + 'px';
                  });
              }

              function getFirstRuledLineY(height) {
                  const lineCount = parseInt(pageSettings.ruledLineCount) || 25;
                  const totalSlots = lineCount + 4;
                  return 2 * (height / totalSlots);
              }

              function renderBackgroundPattern(ctx, width, height) {
                  ctx.fillStyle = pageSettings.bgColor || '#ffffff';
                  ctx.fillRect(0, 0, width, height);

                  if (pageSettings.bgImageDataUrl) {
                      const img = new Image();
                      img.src = pageSettings.bgImageDataUrl;
                      img.onload = () => {
                          ctx.drawImage(img, 0, 0, width, height);
                      };
                  }

                  const pattern = pageSettings.background;
                  ctx.strokeStyle = '#e5e5e5';
                  ctx.lineWidth = 1;
                  ctx.beginPath();

                  if (pattern === 'bg-ruled') {
                      const lineCount = parseInt(pageSettings.ruledLineCount) || 25;
                      const totalSlots = lineCount + 4;
                      const stepY = height / totalSlots;
                      ctx.strokeStyle = '#a2bfe0';
                      for (let k = 2; k <= totalSlots - 2; k++) {
                          ctx.moveTo(0, k * stepY); ctx.lineTo(width, k * stepY);
                      }
                  } else if (pattern === 'bg-grid') {
                      for (let x = 25; x < width; x += 25) { ctx.moveTo(x, 0); ctx.lineTo(x, height); }
                      for (let y = 25; y < height; y += 25) { ctx.moveTo(0, y); ctx.lineTo(width, y); }
                  } else if (pattern === 'bg-dotted') {
                      ctx.fillStyle = '#aaaaaa';
                      for (let x = 25; x < width; x += 25) {
                          for (let y = 25; y < height; y += 25) { ctx.fillRect(x, y, 1.5, 1.5); }
                      }
                  }
                  ctx.stroke();
              }

              function renderMarginGuideOverlay(overlayElem, dims) {
                  overlayElem.innerHTML = '';
                  const type = pageSettings.pageMarginGuide;
                  if (type === 'none') return;

                  const svg = document.createElementNS('http://www.w3.org/2000/svg', 'svg');
                  svg.setAttribute('width', '100%'); svg.setAttribute('height', '100%');
                  svg.style.position = 'absolute';

                  if (type === 'notebook' || type === 'header') {
                      const redLine = document.createElementNS('http://www.w3.org/2000/svg', 'line');
                      redLine.setAttribute('x1', '70'); redLine.setAttribute('y1', '0');
                      redLine.setAttribute('x2', '70'); redLine.setAttribute('y2', dims.height);
                      redLine.setAttribute('stroke', '#ef4444'); redLine.setAttribute('stroke-width', '2');
                      svg.appendChild(redLine);
                  }

                  if (type === 'header') {
                      const headerY = pageSettings.background === 'bg-ruled' ? getFirstRuledLineY(dims.height) : 80;
                      const headerLine = document.createElementNS('http://www.w3.org/2000/svg', 'line');
                      headerLine.setAttribute('x1', '0'); headerLine.setAttribute('y1', headerY);
                      headerLine.setAttribute('x2', dims.width); headerLine.setAttribute('y2', headerY);
                      headerLine.setAttribute('stroke', '#93c5fd'); headerLine.setAttribute('stroke-width', '2');
                      svg.appendChild(headerLine);
                  }

                  if (type === 'box') {
                      const rect = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
                      rect.setAttribute('x', '40'); rect.setAttribute('y', '40');
                      rect.setAttribute('width', dims.width - 80); rect.setAttribute('height', dims.height - 80);
                      rect.setAttribute('fill', 'none'); rect.setAttribute('stroke', '#ef4444');
                      rect.setAttribute('stroke-width', '1.5'); rect.setAttribute('stroke-dasharray', '4 4');
                      svg.appendChild(rect);
                  }

                  overlayElem.appendChild(svg);
              }

              function drawMarginGuidesOnCanvas(ctx, dims) {
                  const type = pageSettings.pageMarginGuide;
                  if (type === 'none') return;

                  ctx.save();
                  if (type === 'notebook' || type === 'header') {
                      ctx.strokeStyle = '#ef4444'; ctx.lineWidth = 2;
                      ctx.beginPath(); ctx.moveTo(70, 0); ctx.lineTo(70, dims.height); ctx.stroke();
                  }
                  if (type === 'header') {
                      const headerY = pageSettings.background === 'bg-ruled' ? getFirstRuledLineY(dims.height) : 80;
                      ctx.strokeStyle = '#93c5fd'; ctx.lineWidth = 2;
                      ctx.beginPath(); ctx.moveTo(0, headerY); ctx.lineTo(dims.width, headerY); ctx.stroke();
                  }
                  if (type === 'box') {
                      ctx.strokeStyle = '#ef4444'; ctx.lineWidth = 1.5; ctx.setLineDash([4, 4]);
                      ctx.strokeRect(40, 40, dims.width - 80, dims.height - 80);
                  }
                  ctx.restore();
              }

              function updateAllPagesLayout() {
                  pages.forEach((_, idx) => {
                      const tempCanvas = document.createElement('canvas');
                      tempCanvas.width = pages[idx].canvas.width;
                      tempCanvas.height = pages[idx].canvas.height;
                      tempCanvas.getContext('2d').drawImage(pages[idx].canvas, 0, 0);

                      applyLayoutToPage(idx);
                      pages[idx].ctx.drawImage(tempCanvas, 0, 0, tempCanvas.width / RENDER_SCALE, tempCanvas.height / RENDER_SCALE);
                  });
                  centerPages();
              }

              function getStrokeStyle(ctx, p1, p2) {
                  if (useGradient) {
                      const grad = ctx.createLinearGradient(p1.x, p1.y, p2 ? p2.x : p1.x + 50, p2 ? p2.y : p1.y + 50);
                      grad.addColorStop(0, gradientPrimary);
                      grad.addColorStop(1, gradientSecondary);
                      return grad;
                  }
                  return currentColor;
              }

              function configurePenStyle(ctx, style, color, width, p1, p2) {
                  ctx.shadowBlur = 0; ctx.shadowColor = 'transparent';
                  if (activeTool === 'eraser') {
                      ctx.globalCompositeOperation = 'destination-out';
                      ctx.strokeStyle = '#000000'; 
                      ctx.lineWidth = lastEraserSize * 2.5;
                      ctx.globalAlpha = 1.0; 
                      ctx.lineCap = 'round'; 
                      ctx.lineJoin = 'round';
                      return;
                  }
                  ctx.globalCompositeOperation = 'source-over';
                  ctx.strokeStyle = getStrokeStyle(ctx, p1 || {x:0,y:0}, p2); ctx.lineCap = 'round'; ctx.lineJoin = 'round';

                  switch(style) {
                      case 'fineliner': ctx.lineWidth = width * 0.8; ctx.globalAlpha = 1.0; break;
                      case 'ballpoint': ctx.lineWidth = width * 0.95; ctx.globalAlpha = 0.9; break;
                      case 'fountain': ctx.lineWidth = width * 1.0; ctx.globalAlpha = 1.0; break;
                      case 'sharptaper': ctx.lineWidth = width * 1.0; ctx.globalAlpha = 1.0; break;
                      case 'calligraphy': ctx.lineWidth = width * 2.2; ctx.lineCap = 'square'; ctx.globalAlpha = 0.95; break;
                      case 'brush': ctx.lineWidth = width * 1.8; ctx.globalAlpha = 0.88; break;
                      case 'marker': ctx.lineWidth = width * 3.0; ctx.lineCap = 'butt'; ctx.globalAlpha = 0.92; break;
                      case 'pencil': ctx.lineWidth = width * 0.75; ctx.globalAlpha = 0.55; break;
                      case 'neon': ctx.lineWidth = width * 1.2; ctx.globalAlpha = 1.0; ctx.shadowColor = useGradient ? gradientPrimary : color; ctx.shadowBlur = 12; break;
                      case 'charcoal': ctx.lineWidth = width * 2.0; ctx.globalAlpha = 0.65; break;
                      case 'highlighter':
                          ctx.globalCompositeOperation = 'destination-over';
                          ctx.lineWidth = width * 4.5;
                          ctx.lineCap = 'square'; ctx.lineJoin = 'miter'; ctx.globalAlpha = 0.45;
                          break;
                  }
              }

              function catmullRomSpline(p0, p1, p2, p3, t) {
                  const v0 = (p2.x - p0.x) * 0.5, v1 = (p3.x - p1.x) * 0.5;
                  const t2 = t * t, t3 = t2 * t;
                  const x = (2 * p1.x - 2 * p2.x + v0 + v1) * t3 + (-3 * p1.x + 3 * p2.x - 2 * v0 - v1) * t2 + v0 * t + p1.x;

                  const wy0 = (p2.y - p0.y) * 0.5, wy1 = (p3.y - p1.y) * 0.5;
                  const y = (2 * p1.y - 2 * p2.y + wy0 + wy1) * t3 + (-3 * p1.y + 3 * p2.y - 2 * wy0 - wy1) * t2 + wy0 * t + p1.y;

                  return { x, y };
              }

              function renderSharpTaperStroke(ctx, points, customSize = strokeSize) {
                  const baseW = customSize * 1.2;

                  if (points.length === 1) {
                      ctx.fillStyle = getStrokeStyle(ctx, points[0]);
                      ctx.beginPath();
                      ctx.arc(points[0].x, points[0].y, 0.5, 0, Math.PI * 2);
                      ctx.fill();
                      return;
                  }

                  let pts = points;
                  if (pts.length >= 2) {
                      let smoothPts = [pts[0]];
                      let extendedPts = [pts[0], ...pts, pts[pts.length - 1]];

                      for (let i = 1; i < extendedPts.length - 2; i++) {
                          let p0 = extendedPts[i - 1];
                          let p1 = extendedPts[i];
                          let p2 = extendedPts[i + 1];
                          let p3 = extendedPts[i + 2];
                          
                          let dist = Math.hypot(p2.x - p1.x, p2.y - p1.y);
                          let steps = Math.max(3, Math.ceil(dist / 2));

                          for (let k = 1; k <= steps; k++) {
                              smoothPts.push(catmullRomSpline(p0, p1, p2, p3, k / steps));
                          }
                      }
                      pts = smoothPts;
                  }

                  let totalDist = 0;
                  let cumDists = [0];
                  for (let i = 1; i < pts.length; i++) {
                      totalDist += Math.hypot(pts[i].x - pts[i - 1].x, pts[i].y - pts[i - 1].y);
                      cumDists.push(totalDist);
                  }

                  if (totalDist === 0) return;

                  let leftSide = [];
                  let rightSide = [];

                  for (let i = 0; i < pts.length; i++) {
                      let progress = cumDists[i] / totalDist;
                      let taperFactor = Math.sin(progress * Math.PI);
                      let currentWidth = baseW * Math.pow(taperFactor, 0.8);

                      let dx, dy;
                      if (i === 0) {
                          dx = pts[1].x - pts[0].x;
                          dy = pts[1].y - pts[0].y;
                      } else if (i === pts.length - 1) {
                          dx = pts[i].x - pts[i - 1].x;
                          dy = pts[i].y - pts[i - 1].y;
                      } else {
                          dx = pts[i + 1].x - pts[i - 1].x;
                          dy = pts[i + 1].y - pts[i - 1].y;
                      }

                      let len = Math.hypot(dx, dy) || 1;
                      let nx = -dy / len;
                      let ny = dx / len;

                      let r = currentWidth / 2;

                      leftSide.push({
                          x: pts[i].x + nx * r,
                          y: pts[i].y + ny * r
                      });
                      rightSide.push({
                          x: pts[i].x - nx * r,
                          y: pts[i].y - ny * r
                      });
                  }

                  ctx.fillStyle = getStrokeStyle(ctx, pts[0], pts[pts.length - 1]);
                  ctx.beginPath();
                  ctx.moveTo(leftSide[0].x, leftSide[0].y);

                  for (let i = 1; i < leftSide.length - 1; i++) {
                      let xc = (leftSide[i].x + leftSide[i + 1].x) / 2;
                      let yc = (leftSide[i].y + leftSide[i + 1].y) / 2;
                      ctx.quadraticCurveTo(leftSide[i].x, leftSide[i].y, xc, yc);
                  }
                  if (leftSide.length > 1) {
                      let lastL = leftSide[leftSide.length - 1];
                      ctx.lineTo(lastL.x, lastL.y);
                  }

                  for (let i = rightSide.length - 1; i > 0; i--) {
                      let xc = (rightSide[i].x + rightSide[i - 1].x) / 2;
                      let yc = (rightSide[i].y + rightSide[i - 1].y) / 2;
                      ctx.quadraticCurveTo(rightSide[i].x, rightSide[i].y, xc, yc);
                  }
                  ctx.lineTo(rightSide[0].x, rightSide[0].y);

                  ctx.closePath();
                  ctx.fill();
              }

              function renderSmoothStroke(ctx, points, customSize = strokeSize) {
                  if (!points || points.length === 0) return;

                  if (currentPenStyle === 'sharptaper' && activeTool !== 'eraser') {
                      ctx.globalCompositeOperation = 'source-over';
                      ctx.globalAlpha = 1.0;
                      renderSharpTaperStroke(ctx, points, customSize);
                      return;
                  }

                  if (currentPenStyle === 'fountain' && activeTool !== 'eraser') {
                      ctx.globalCompositeOperation = 'source-over';
                      const style = getStrokeStyle(ctx, points[0], points[points.length - 1]);
                      ctx.strokeStyle = style;
                      ctx.fillStyle = style;
                      ctx.lineCap = 'round';
                      ctx.lineJoin = 'round';
                      ctx.globalAlpha = 1.0;

                      const baseW = customSize * 1.0;

                      if (points.length === 1) {
                          ctx.beginPath();
                          ctx.arc(points[0].x, points[0].y, baseW / 2, 0, Math.PI * 2);
                          ctx.fill();
                          return;
                      }

                      let totalLength = 0;
                      const dists = [0];
                      for (let i = 1; i < points.length; i++) {
                          const d = Math.hypot(points[i].x - points[i-1].x, points[i].y - points[i-1].y);
                          totalLength += d;
                          dists.push(totalLength);
                      }

                      for (let i = 1; i < points.length; i++) {
                          const p1 = points[i - 1];
                          const p2 = points[i];

                          const progress1 = totalLength > 0 ? dists[i - 1] / totalLength : 0;
                          const progress2 = totalLength > 0 ? dists[i] / totalLength : 1;

                          const minW = baseW * 1.0;
                          const maxW = baseW * 2.5;
                          const w1 = minW + (maxW - minW) * progress1;
                          const w2 = minW + (maxW - minW) * progress2;

                          const currentW = (w1 + w2) / 2;
                          ctx.lineWidth = Math.max(0.2, currentW);
                          ctx.beginPath();

                          if (points.length === 2) {
                              ctx.moveTo(p1.x, p1.y);
                              ctx.lineTo(p2.x, p2.y);
                          } else if (i === 1) {
                              const midX = (p1.x + p2.x) / 2;
                              const midY = (p1.y + p2.y) / 2;
                              ctx.moveTo(p1.x, p1.y);
                              ctx.lineTo(midX, midY);
                          } else if (i === points.length - 1) {
                              const prevMidX = (points[i - 2].x + p1.x) / 2;
                              const prevMidY = (points[i - 2].y + p1.y) / 2;
                              ctx.moveTo(prevMidX, prevMidY);
                              ctx.quadraticCurveTo(p1.x, p1.y, p2.x, p2.y);
                          } else {
                              const prevMidX = (points[i - 2].x + p1.x) / 2;
                              const prevMidY = (points[i - 2].y + p1.y) / 2;
                              const midX = (p1.x + p2.x) / 2;
                              const midY = (p1.y + p2.y) / 2;
                              ctx.moveTo(prevMidX, prevMidY);
                              ctx.quadraticCurveTo(p1.x, p1.y, midX, midY);
                          }
                          ctx.stroke();
                      }
                      return;
                  }

                  configurePenStyle(ctx, currentPenStyle, currentColor, customSize, points[0], points[points.length - 1]);

                  if (points.length === 1) {
                      ctx.beginPath();
                      ctx.arc(points[0].x, points[0].y, ctx.lineWidth / 2, 0, Math.PI * 2);
                      ctx.fillStyle = ctx.strokeStyle; 
                      ctx.fill();
                      return;
                  }

                  if (points.length === 2) {
                      ctx.beginPath();
                      ctx.moveTo(points[0].x, points[0].y);
                      ctx.lineTo(points[1].x, points[1].y);
                      ctx.stroke();
                      return;
                  }

                  ctx.beginPath();
                  const firstMidX = (points[0].x + points[1].x) / 2;
                  const firstMidY = (points[0].y + points[1].y) / 2;
                  ctx.moveTo(points[0].x, points[0].y);
                  ctx.lineTo(firstMidX, firstMidY);

                  for (let i = 1; i < points.length - 1; i++) {
                      const xc = (points[i].x + points[i + 1].x) / 2;
                      const yc = (points[i].y + points[i + 1].y) / 2;
                      ctx.quadraticCurveTo(points[i].x, points[i].y, xc, yc);
                  }

                  const lastIdx = points.length - 1;
                  ctx.lineTo(points[lastIdx].x, points[lastIdx].y);
                  ctx.stroke();
              }

              function getCanvasCoords(clientX, clientY, canvas) {
                  const rect = canvas.getBoundingClientRect();
                  return { x: (clientX - rect.left) / scale, y: (clientY - rect.top) / scale, time: Date.now() };
              }

              window.addEventListener('pointerdown', (e) => {
                  if (!studioView.contains(e.target)) return;
                  if (e.target.closest('#toolbar') || e.target.closest('.popover-menu') || e.target.closest('#lassoActionMenu') || e.target.closest('.modal-overlay') || e.target.closest('.page-branding-bar')) return;

                  const pIdx = pointers.findIndex(p => p.pointerId === e.pointerId);
                  if (pIdx === -1) pointers.push(e);
                  else pointers[pIdx] = e;

                  if (pointers.length === 1) {
                      const card = e.target.closest('.page-card');
                      const isImageInteract = e.target.closest('.image-box');

                      if (!card || ((activeTool === 'editImage' || activeTool === 'writeInsideImage') && !isImageInteract)) {
                          clearLassoBox();
                          isPanningCanvas = true;
                          lastPanPos = { x: e.clientX, y: e.clientY };
                          return;
                      }

                      if (activeTool === 'editImage' || activeTool === 'writeInsideImage') return;

                      if (card) {
                          activePageIndex = parseInt(card.dataset.index);
                          const page = pages[activePageIndex];

                          if (activeTool === 'lasso') {
                              clearLassoBox();
                              isLassoing = true;
                              const coords = getCanvasCoords(e.clientX, e.clientY, page.canvas);
                              lassoStartPos = coords;
                              currentLassoRect = { pageIdx: activePageIndex, x: coords.x, y: coords.y, width: 0, height: 0 };
                              return;
                          }

                          if (activeTool === 'text') {
                              clearLassoBox();
                              const coords = getCanvasCoords(e.clientX, e.clientY, page.canvas);
                              createTextBox(activePageIndex, coords.x, coords.y);
                              return;
                          }

                          clearLassoBox();
                          isDrawing = true;
                          hasZoomedOrPanned = false;
                          preStrokeSnapshot = page.ctx.getImageData(0, 0, page.canvas.width, page.canvas.height);

                          const coords = getCanvasCoords(e.clientX, e.clientY, page.canvas);
                          currentStrokePoints = [coords];
                          renderSmoothStroke(page.ctx, currentStrokePoints);
                      }
                  } 
                  else if (pointers.length === 2) {
                      isDrawing = false;
                      isLassoing = false;
                      isPanningCanvas = false;
                      hasZoomedOrPanned = true;
                      currentStrokePoints = [];

                      if (activePageIndex !== null && preStrokeSnapshot) {
                          pages[activePageIndex].ctx.putImageData(preStrokeSnapshot, 0, 0);
                      }
                      preStrokeSnapshot = null;

                      initialZoomDist = Math.hypot(pointers[0].clientX - pointers[1].clientX, pointers[0].clientY - pointers[1].clientY);
                      initialScale = scale;
                      initialPanX = panX; initialPanY = panY;
                      initialCenter = { x: (pointers[0].clientX + pointers[1].clientX) / 2, y: (pointers[0].clientY + pointers[1].clientY) / 2 };
                  }
              });

              window.addEventListener('pointermove', (e) => {
                  const index = pointers.findIndex(p => p.pointerId === e.pointerId);
                  if (index !== -1) pointers[index] = e;

                  if (pointers.length === 1 && isPanningCanvas) {
                      hasZoomedOrPanned = true;
                      const dx = e.clientX - lastPanPos.x;
                      const dy = e.clientY - lastPanPos.y;
                      panX += dx;
                      panY += dy;
                      lastPanPos = { x: e.clientX, y: e.clientY };
                      updateTransform();
                      return;
                  }

                  if (pointers.length === 1 && isLassoing && activePageIndex !== null) {
                      const page = pages[activePageIndex];
                      const coords = getCanvasCoords(e.clientX, e.clientY, page.canvas);
                      
                      const minX = Math.min(lassoStartPos.x, coords.x);
                      const minY = Math.min(lassoStartPos.y, coords.y);
                      const width = Math.abs(coords.x - lassoStartPos.x);
                      const height = Math.abs(coords.y - lassoStartPos.y);

                      currentLassoRect = { pageIdx: activePageIndex, x: minX, y: minY, width, height };
                      updateLassoOverlayPosition();
                  }
                  else if (pointers.length === 1 && isDrawing && !hasZoomedOrPanned && activePageIndex !== null) {
                      const page = pages[activePageIndex];
                      const coords = getCanvasCoords(e.clientX, e.clientY, page.canvas);
                      currentStrokePoints.push(coords);

                      if (preStrokeSnapshot) {
                          page.ctx.putImageData(preStrokeSnapshot, 0, 0);
                      }
                      renderSmoothStroke(page.ctx, currentStrokePoints);

                  } else if (pointers.length === 2) {
                      hasZoomedOrPanned = true;
                      const currentCenter = { x: (pointers[0].clientX + pointers[1].clientX) / 2, y: (pointers[0].clientY + pointers[1].clientY) / 2 };
                      const currentDist = Math.hypot(pointers[0].clientX - pointers[1].clientX, pointers[0].clientY - pointers[1].clientY);

                      const newScale = Math.min(Math.max(initialScale * (currentDist / initialZoomDist), 0.2), 4.5);
                      scale = newScale;
                      panX = currentCenter.x - ((initialCenter.x - initialPanX) * (newScale / initialScale));
                      panY = currentCenter.y - ((initialCenter.y - initialPanY) * (newScale / initialScale));

                      updateTransform();
                  }
              });

              function handlePointerUp(e) {
                  pointers = pointers.filter(p => p.pointerId !== e.pointerId);
                  if (pointers.length < 2) initialZoomDist = null;

                  if (pointers.length === 0) {
                      if (isPanningCanvas) {
                          isPanningCanvas = false;
                      }
                      else if (isLassoing) {
                          isLassoing = false;
                          if (currentLassoRect && (currentLassoRect.width < 10 || currentLassoRect.height < 10)) {
                              clearLassoBox();
                          } else {
                              showLassoActionMenu();
                          }
                      }
                      isDrawing = false;
                      hasZoomedOrPanned = false;
                      currentStrokePoints = [];
                      activePageIndex = null;
                      preStrokeSnapshot = null;
                  }
              }

              window.addEventListener('pointerup', handlePointerUp);
              window.addEventListener('pointercancel', handlePointerUp);

              viewport.addEventListener('wheel', (e) => {
                  e.preventDefault();
                  hasZoomedOrPanned = true;
                  if (e.ctrlKey) {
                      const newScale = Math.min(Math.max(scale * (e.deltaY < 0 ? 1.08 : 0.92), 0.2), 4.5);
                      panX = e.clientX - (e.clientX - panX) * (newScale / scale);
                      panY = e.clientY - (e.clientY - panY) * (newScale / scale);
                      scale = newScale;
                  } else {
                      panX -= e.deltaX; panY -= e.deltaY;
                  }
                  updateTransform();
              }, { passive: false });

              function hexToRgb(hex) {
                  let c = hex.replace('#', '');
                  if (c.length === 3) c = c.split('').map(x => x + x).join('');
                  const num = parseInt(c, 16);
                  return { r: (num >> 16) & 255, g: (num >> 8) & 255, b: num & 255 };
              }

              document.getElementById('lassoColorPicker').oninput = (e) => {
                  if (!currentLassoRect) return;
                  const page = pages[currentLassoRect.pageIdx];
                  const targetColor = hexToRgb(e.target.value);

                  const rx = currentLassoRect.x * RENDER_SCALE;
                  const ry = currentLassoRect.y * RENDER_SCALE;
                  const rw = currentLassoRect.width * RENDER_SCALE;
                  const rh = currentLassoRect.height * RENDER_SCALE;

                  const imgData = page.ctx.getImageData(rx, ry, rw, rh);
                  const data = imgData.data;

                  for (let i = 0; i < data.length; i += 4) {
                      if (data[i + 3] > 20) {
                          data[i] = targetColor.r;
                          data[i + 1] = targetColor.g;
                          data[i + 2] = targetColor.b;
                      }
                  }
                  page.ctx.putImageData(imgData, rx, ry);
              };

              function scaleLassoContent(scaleFactor) {
                  if (!currentLassoRect) return;
                  const page = pages[currentLassoRect.pageIdx];

                  if (currentLassoTargetImage) {
                      let currentW = parseFloat(currentLassoTargetImage.elem.style.width) || currentLassoRect.width;
                      let currentH = parseFloat(currentLassoTargetImage.elem.style.height) || currentLassoRect.height;

                      let newW = currentW * scaleFactor;
                      let newH = currentH * scaleFactor;

                      currentLassoTargetImage.elem.style.width = newW + 'px';
                      currentLassoTargetImage.elem.style.height = newH + 'px';

                      currentLassoRect.width = newW;
                      currentLassoRect.height = newH;

                      updateLassoOverlayPosition();
                      showLassoActionMenu();
                      return;
                  }

                  const rx = currentLassoRect.x * RENDER_SCALE;
                  const ry = currentLassoRect.y * RENDER_SCALE;
                  const rw = currentLassoRect.width * RENDER_SCALE;
                  const rh = currentLassoRect.height * RENDER_SCALE;

                  if (rw <= 0 || rh <= 0) return;

                  const tempCanvas = document.createElement('canvas');
                  tempCanvas.width = rw; tempCanvas.height = rh;
                  const tempCtx = tempCanvas.getContext('2d');
                  tempCtx.putImageData(page.ctx.getImageData(rx, ry, rw, rh), 0, 0);

                  page.ctx.clearRect(rx, ry, rw, rh);

                  const newW = rw * scaleFactor;
                  const newH = rh * scaleFactor;

                  page.ctx.save();
                  page.ctx.drawImage(tempCanvas, 0, 0, rw, rh, rx, ry, newW, newH);
                  page.ctx.restore();

                  currentLassoRect.width *= scaleFactor;
                  currentLassoRect.height *= scaleFactor;

                  page.textBoxes.forEach(tb => {
                      const tbLeft = parseFloat(tb.elem.style.left) || 0;
                      const tbTop = parseFloat(tb.elem.style.top) || 0;
                      if (tbLeft >= currentLassoRect.x && tbLeft <= currentLassoRect.x + currentLassoRect.width &&
                          tbTop >= currentLassoRect.y && tbTop <= currentLassoRect.y + currentLassoRect.height) {
                          let curW = parseFloat(tb.elem.style.width) || tb.elem.offsetWidth;
                          tb.elem.style.width = (curW * scaleFactor) + 'px';
                      }
                  });

                  updateLassoOverlayPosition();
                  showLassoActionMenu();
              }

              function extractTransparentLassoImage() {
                  if (!currentLassoRect) return null;
                  const page = pages[currentLassoRect.pageIdx];

                  const rx = currentLassoRect.x * RENDER_SCALE;
                  const ry = currentLassoRect.y * RENDER_SCALE;
                  const rw = currentLassoRect.width * RENDER_SCALE;
                  const rh = currentLassoRect.height * RENDER_SCALE;

                  if (rw <= 0 || rh <= 0) return null;

                  const cropCanvas = document.createElement('canvas');
                  cropCanvas.width = rw; cropCanvas.height = rh;
                  const cropCtx = cropCanvas.getContext('2d');

                  cropCtx.drawImage(page.canvas, rx, ry, rw, rh, 0, 0, rw, rh);

                  const imgData = cropCtx.getImageData(0, 0, rw, rh);
                  const data = imgData.data;

                  for (let i = 0; i < data.length; i += 4) {
                      const r = data[i], g = data[i + 1], b = data[i + 2];
                      if (r > 235 && g > 235 && b > 235) {
                          data[i + 3] = 0;
                      }
                  }
                  cropCtx.putImageData(imgData, 0, 0);
                  return cropCanvas.toDataURL('image/png');
              }

              function createTextBox(pageIdx, x, y, initialText = '', initialColor = null, initialWidth = 220) {
                  const page = pages[pageIdx];
                  const dims = getPageDimensions();
                  const ruleH = getRuledLineHeight();

                  let safeX = Math.max(0, Math.min(x, dims.width - 60));
                  let safeY = Math.max(0, Math.min(y, dims.height - 30));

                  const wrapper = document.createElement('div');
                  wrapper.className = 'textbox-wrapper';
                  wrapper.style.left = safeX + 'px'; 
                  wrapper.style.top = safeY + 'px';
                  
                  let maxAvailableWidth = dims.width - safeX;
                  let safeW = Math.min(initialWidth, maxAvailableWidth);
                  wrapper.style.width = safeW + 'px';

                  const deleteBtn = document.createElement('div');
                  deleteBtn.className = 'textbox-delete-btn';
                  deleteBtn.innerText = '✕';

                  const resizeHandle = document.createElement('div');
                  resizeHandle.className = 'textbox-resize-handle';

                  const content = document.createElement('div');
                  content.className = 'textbox-content';
                  content.contentEditable = 'false';
                  content.innerText = initialText;

                  if (useGradient) {
                      content.style.background = `linear-gradient(45deg, ${gradientPrimary}, ${gradientSecondary})`;
                      content.style.webkitBackgroundClip = 'text';
                      content.style.webkitTextFillColor = 'transparent';
                  } else {
                      content.style.color = initialColor || currentColor;
                  }

                  content.style.fontSize = (ruleH * 0.72) + 'px';
                  content.style.lineHeight = ruleH + 'px';
                  content.style.minHeight = ruleH + 'px';

                  wrapper.appendChild(deleteBtn);
                  wrapper.appendChild(resizeHandle);
                  wrapper.appendChild(content);
                  page.card.appendChild(wrapper);

                  if (!initialText) {
                      enableEditing();
                  }

                  let isDragging = false, startLeft = safeX, startTop = safeY, startPointerX = 0, startPointerY = 0;
                  let isResizing = false, resizeStartX = 0, startWidth = 0;
                  let lastTap = 0;

                  function enableEditing() {
                      wrapper.classList.add('editing');
                      content.contentEditable = 'true';
                      content.focus();
                  }

                  function disableEditing() {
                      wrapper.classList.remove('editing');
                      content.contentEditable = 'false';
                  }

                  wrapper.addEventListener('pointerdown', (e) => {
                      if (e.target === deleteBtn || e.target === resizeHandle) return;
                      if (content.isContentEditable) return;

                      e.stopPropagation();
                      const now = Date.now();

                      if (now - lastTap < 300) {
                          enableEditing();
                          lastTap = 0;
                          return;
                      }
                      lastTap = now;

                      isDragging = true;
                      startLeft = parseFloat(wrapper.style.left) || safeX;
                      startTop = parseFloat(wrapper.style.top) || safeY;
                      startPointerX = e.clientX;
                      startPointerY = e.clientY;
                      wrapper.setPointerCapture(e.pointerId);
                  });

                  wrapper.addEventListener('pointermove', (e) => {
                      if (!isDragging) return;
                      let dx = (e.clientX - startPointerX) / scale;
                      let dy = (e.clientY - startPointerY) / scale;

                      wrapper.style.left = (startLeft + dx) + 'px';
                      wrapper.style.top = (startTop + dy) + 'px';
                  });

                  wrapper.addEventListener('pointerup', (e) => { 
                      if (isDragging) {
                          isDragging = false; 
                          try { wrapper.releasePointerCapture(e.pointerId); } catch(err){}
                          reassignElementToPage(wrapper, 'text');
                      }
                  });

                  resizeHandle.addEventListener('pointerdown', (e) => {
                      e.stopPropagation();
                      isResizing = true;
                      resizeStartX = e.clientX;
                      startWidth = wrapper.offsetWidth;
                      resizeHandle.setPointerCapture(e.pointerId);
                  });

                  resizeHandle.addEventListener('pointermove', (e) => {
                      if (!isResizing) return;
                      const dw = (e.clientX - resizeStartX) / scale;
                      let newWidth = startWidth + dw;

                      let currentLeft = parseFloat(wrapper.style.left) || 0;
                      let maxWidth = dims.width - currentLeft;
                      let finalWidth = Math.max(80, Math.min(newWidth, maxWidth));

                      wrapper.style.width = finalWidth + 'px';
                  });

                  resizeHandle.addEventListener('pointerup', (e) => {
                      isResizing = false;
                      try { resizeHandle.releasePointerCapture(e.pointerId); } catch(err){}
                  });

                  content.addEventListener('blur', () => disableEditing());

                  deleteBtn.onclick = (e) => {
                      e.stopPropagation();
                      wrapper.remove();
                      pages.forEach(p => p.textBoxes = p.textBoxes.filter(t => t.elem !== wrapper));
                  };

                  page.textBoxes.push({ elem: wrapper, contentElem: content });
              }

              function insertImageToPage(pageIdx, imgObj, src, savedState = null) {
                  const page = pages[pageIdx];
                  let initW = savedState ? savedState.width : Math.min(imgObj.width, 300);
                  let initH = savedState ? savedState.height : (imgObj.height / imgObj.width) * initW;
                  
                  const existingCount = page.imageBoxes.length;
                  let initTop = savedState ? savedState.top : 60 + (existingCount * 80);
                  let initLeft = savedState ? savedState.left : 60;

                  const box = document.createElement('div');
                  box.className = 'image-box';
                  box.style.left = initLeft + 'px'; box.style.top = initTop + 'px';
                  box.style.width = initW + 'px'; box.style.height = initH + 'px';

                  const deleteBtn = document.createElement('div');
                  deleteBtn.className = 'image-delete-btn'; deleteBtn.innerText = '✕';

                  const container = document.createElement('div');
                  container.className = 'image-container';
                  const img = document.createElement('img');
                  img.src = src;

                  const inkCanvas = document.createElement('canvas');
                  inkCanvas.className = 'image-ink-canvas';
                  
                  inkCanvas.width = 1000;
                  inkCanvas.height = 1000 * (initH / initW);
                  const inkCtx = inkCanvas.getContext('2d');

                  container.appendChild(img);
                  container.appendChild(inkCanvas);

                  const resizeHandle = document.createElement('div');
                  resizeHandle.className = 'resize-handle';

                  box.appendChild(deleteBtn);
                  box.appendChild(container); 
                  box.appendChild(resizeHandle);
                  page.card.appendChild(box);

                  if (savedState && savedState.imageInkData) {
                      const loadedInk = new Image();
                      loadedInk.src = savedState.imageInkData;
                      loadedInk.onload = () => {
                          inkCtx.drawImage(loadedInk, 0, 0, inkCanvas.width, inkCanvas.height);
                      };
                  }

                  const imageEntry = { elem: box, imgElem: img, inkCanvas: inkCanvas, inkCtx: inkCtx };

                  let isWritingOnImage = false;
                  let imgStrokePoints = [];
                  let imgSnapshot = null;

                  function getInnerImageCoords(e) {
                      const rect = inkCanvas.getBoundingClientRect();
                      return {
                          x: ((e.clientX - rect.left) / rect.width) * inkCanvas.width,
                          y: ((e.clientY - rect.top) / rect.height) * inkCanvas.height
                      };
                  }

                  inkCanvas.addEventListener('pointerdown', (e) => {
                      if (activeTool !== 'writeInsideImage' || pointers.length > 1) return;
                      e.stopPropagation();
                      isWritingOnImage = true;
                      inkCanvas.setPointerCapture(e.pointerId);

                      imgSnapshot = inkCtx.getImageData(0, 0, inkCanvas.width, inkCanvas.height);
                      const coords = getInnerImageCoords(e);
                      imgStrokePoints = [coords];

                      const scaleFactor = inkCanvas.width / box.offsetWidth;
                      renderSmoothStroke(inkCtx, imgStrokePoints, strokeSize * scaleFactor);
                  });

                  inkCanvas.addEventListener('pointermove', (e) => {
                      if (!isWritingOnImage || pointers.length > 1) return;
                      e.stopPropagation();
                      const coords = getInnerImageCoords(e);
                      imgStrokePoints.push(coords);

                      if (imgSnapshot) {
                          inkCtx.putImageData(imgSnapshot, 0, 0);
                      }
                      const scaleFactor = inkCanvas.width / box.offsetWidth;
                      renderSmoothStroke(inkCtx, imgStrokePoints, strokeSize * scaleFactor);
                  });

                  inkCanvas.addEventListener('pointerup', (e) => {
                      if (!isWritingOnImage) return;
                      e.stopPropagation();
                      isWritingOnImage = false;
                      try { inkCanvas.releasePointerCapture(e.pointerId); } catch(err){}
                      imgStrokePoints = [];
                      imgSnapshot = null;
                  });

                  let isDragging = false, startLeft = initLeft, startTop = initTop, startPointerX = 0, startPointerY = 0;
                  box.addEventListener('pointerdown', (e) => {
                      if (activeTool !== 'editImage' || pointers.length > 1) return;
                      if (e.target === resizeHandle || e.target === deleteBtn) return;
                      e.stopPropagation(); 
                      isDragging = true;
                      startLeft = parseFloat(box.style.left) || initLeft;
                      startTop = parseFloat(box.style.top) || initTop;
                      startPointerX = e.clientX;
                      startPointerY = e.clientY;
                      box.setPointerCapture(e.pointerId);
                  });

                  box.addEventListener('pointermove', (e) => {
                      if (!isDragging || pointers.length > 1) return;
                      let dx = (e.clientX - startPointerX) / scale;
                      let dy = (e.clientY - startPointerY) / scale;

                      box.style.left = (startLeft + dx) + 'px';
                      box.style.top = (startTop + dy) + 'px';
                  });

                  box.addEventListener('pointerup', (e) => { 
                      if (isDragging) {
                          isDragging = false;

              function generateProceduralSvgPattern(type, color) {
                  const svgHeader = `data:image/svg+xml;utf8,`;
                  let body = '';

                  if (type === 'grid') {
                      body = `<svg xmlns="http://www.w3.org/2000/svg" width="30" height="30"><path d="M 30 0 L 0 0 0 30" fill="none" stroke="%23cbd5e1" stroke-width="1"/></svg>`;
                  } else if (type === 'dots') {
                      body = `<svg xmlns="http://www.w3.org/2000/svg" width="24" height="24"><circle cx="12" cy="12" r="1.5" fill="%2394a3b8"/></svg>`;
                  } else if (type === 'iso') {
                      body = `<svg xmlns="http://www.w3.org/2000/svg" width="30" height="52"><path d="M14 0 L28 8 L28 24 L14 32 L0 24 L0 8 Z M14 32 L28 40 L28 48 M14 32 L0 40 L0 48" fill="none" stroke="%23cbd5e1" stroke-width="1"/></svg>`;
                  } else if (type === 'lines') {
                      body = `<svg xmlns="http://www.w3.org/2000/svg" width="20" height="20"><path d="M0 20 L20 0 Z" fill="none" stroke="%23cbd5e1" stroke-width="1"/></svg>`;
                  } else if (type === 'cross') {
                      body = `<svg xmlns="http://www.w3.org/2000/svg" width="20" height="20"><path d="M0 0 L20 20 M20 0 L0 20" fill="none" stroke="%23e2e8f0" stroke-width="1"/></svg>`;
                  } else if (type === 'waves') {
                      body = `<svg xmlns="http://www.w3.org/2000/svg" width="40" height="20"><path d="M0 10 Q10 0, 20 10 T40 10" fill="none" stroke="%23cbd5e1" stroke-width="1"/></svg>`;
                  } else if (type === 'blueprint') {
                      body = `<svg xmlns="http://www.w3.org/2000/svg" width="40" height="40"><path d="M 40 0 L 0 0 0 40" fill="none" stroke="%233b82f6" stroke-width="0.8" opacity="0.3"/></svg>`;
                  } else if (type === 'honeycomb') {
                      body = `<svg xmlns="http://www.w3.org/2000/svg" width="28" height="48"><path d="M14 0 L28 8 L28 24 L14 32 L0 24 L0 8 Z M14 32 L28 40 L28 48 M14 32 L0 40 L0 48" fill="none" stroke="%23cbd5e1" stroke-width="1"/></svg>`;
                  }

                  return svgHeader + encodeURIComponent(body);
              }

              function makeTextImgBoxInteractive(box) {
                  const dragHandle = box.querySelector('.text-img-drag-handle');
                  const deleteBtn = box.querySelector('.text-img-delete-btn');
                  const resizeHandle = box.querySelector('.text-img-resize-handle');
                  const canvas = box.querySelector('.text-img-ink-canvas');

                  deleteBtn.onclick = (e) => {
                      e.stopPropagation();
                      box.remove();
                  };

                  let isDragging = false, startX = 0, startY = 0, initialLeft = 0, initialTop = 0;

                  const onDragStart = (e) => {
                      e.stopPropagation();
                      if (e.touches && e.touches.length > 0) {
                          startX = e.touches[0].clientX;
                          startY = e.touches[0].clientY;
                      } else {
                          startX = e.clientX;
                          startY = e.clientY;
                      }
                      isDragging = true;
                      initialLeft = parseFloat(box.style.left) || 40;
                      initialTop = parseFloat(box.style.top) || 120;
                      if (e.pointerId && dragHandle.setPointerCapture) {
                          try { dragHandle.setPointerCapture(e.pointerId); } catch(err){}
                      }
                  };

                  const onDragMove = (e) => {
                      if (!isDragging) return;
                      e.preventDefault();
                      e.stopPropagation();
                      let clientX = e.clientX;
                      let clientY = e.clientY;
                      if (e.touches && e.touches.length > 0) {
                          clientX = e.touches[0].clientX;
                          clientY = e.touches[0].clientY;
                      }
                      let dx = clientX - startX;
                      let dy = clientY - startY;
                      box.style.left = (initialLeft + dx) + 'px';
                      box.style.top = (initialTop + dy) + 'px';
                  };

                  const onDragEnd = (e) => {
                      if (isDragging) {
                          isDragging = false;
                          if (e.pointerId && dragHandle.releasePointerCapture) {
                              try { dragHandle.releasePointerCapture(e.pointerId); } catch(err){}
                          }
                      }
                  };

                  dragHandle.addEventListener('pointerdown', onDragStart);
                  dragHandle.addEventListener('pointermove', onDragMove);
                  dragHandle.addEventListener('pointerup', onDragEnd);
                  dragHandle.addEventListener('pointercancel', onDragEnd);

                  dragHandle.addEventListener('touchstart', onDragStart, { passive: false });
                  dragHandle.addEventListener('touchmove', onDragMove, { passive: false });
                  dragHandle.addEventListener('touchend', onDragEnd);

                  let isResizing = false, rStartX = 0, rStartY = 0, startW = 0, startH = 0;

                  const onResizeStart = (e) => {
                      e.stopPropagation();
                      isResizing = true;
                      if (e.touches && e.touches.length > 0) {
                          rStartX = e.touches[0].clientX;
                          rStartY = e.touches[0].clientY;
                      } else {
                          rStartX = e.clientX;
                          rStartY = e.clientY;
                      }
                      startW = box.offsetWidth;
                      startH = box.offsetHeight;
                      if (e.pointerId && resizeHandle.setPointerCapture) {
                          try { resizeHandle.setPointerCapture(e.pointerId); } catch(err){}
                      }
                  };

                  const onResizeMove = (e) => {
                      if (!isResizing) return;
                      e.preventDefault();
                      e.stopPropagation();
                      let clientX = e.clientX;
                      let clientY = e.clientY;
                      if (e.touches && e.touches.length > 0) {
                          clientX = e.touches[0].clientX;
                          clientY = e.touches[0].clientY;
                      }
                      let dw = clientX - rStartX;
                      let dh = clientY - rStartY;
                      box.style.width = Math.max(120, startW + dw) + 'px';
                      box.style.height = Math.max(80, startH + dh) + 'px';
                  };

                  const onResizeEnd = (e) => {
                      if (isResizing) {
                          isResizing = false;
                          if (e.pointerId && resizeHandle.releasePointerCapture) {
                              try { resizeHandle.releasePointerCapture(e.pointerId); } catch(err){}
                          }
                      }
                  };

                  resizeHandle.addEventListener('pointerdown', onResizeStart);
                  resizeHandle.addEventListener('pointermove', onResizeMove);
                  resizeHandle.addEventListener('pointerup', onResizeEnd);
                  resizeHandle.addEventListener('pointercancel', onResizeEnd);

                  resizeHandle.addEventListener('touchstart', onResizeStart, { passive: false });
                  resizeHandle.addEventListener('touchmove', onResizeMove, { passive: false });
                  resizeHandle.addEventListener('touchend', onResizeEnd);

                  if (canvas) {
                      const ctx = canvas.getContext('2d');
                      let isDrawingImg = false;

                      const getEventPos = (ev) => {
                          const rect = canvas.getBoundingClientRect();
                          let clientX = ev.clientX;
                          let clientY = ev.clientY;
                          if (ev.touches && ev.touches.length > 0) {
                              clientX = ev.touches[0].clientX;
                              clientY = ev.touches[0].clientY;
                          }
                          return {
                              x: ((clientX - rect.left) / rect.width) * canvas.width,
                              y: ((clientY - rect.top) / rect.height) * canvas.height
                          };
                      };

                      const onDrawStart = (ev) => {
                          ev.stopPropagation();
                          isDrawingImg = true;
                          ctx.strokeStyle = document.getElementById('textImageInkColorPicker').value;
                          ctx.lineWidth = 3;
                          ctx.lineCap = 'round';
                          const pos = getEventPos(ev);
                          ctx.beginPath();
                          ctx.moveTo(pos.x, pos.y);
                      };

                      const onDrawMove = (ev) => {
                          if (!isDrawingImg) return;
                          ev.stopPropagation();
                          ev.preventDefault();
                          const pos = getEventPos(ev);
                          ctx.lineTo(pos.x, pos.y);
                          ctx.stroke();
                      };

                      const onDrawEnd = () => isDrawingImg = false;

                      canvas.addEventListener('pointerdown', onDrawStart);
                      canvas.addEventListener('pointermove', onDrawMove);
                      canvas.addEventListener('pointerup', onDrawEnd);

                      canvas.addEventListener('touchstart', onDrawStart, { passive: false });
                      canvas.addEventListener('touchmove', onDrawMove, { passive: false });
                      canvas.addEventListener('touchend', onDrawEnd);
                  }
              }

              function renderPageToCanvas(pageIdx) {
                  const page = pages[pageIdx];
                  const dims = getPageDimensions();
                  const ruleH = getRuledLineHeight();
                  const expCanvas = document.createElement('canvas');
                  expCanvas.width = dims.width * 2; expCanvas.height = dims.height * 2;
                  const ctx = expCanvas.getContext('2d'); ctx.scale(2, 2);

                  ctx.drawImage(page.bgCanvas, 0, 0, dims.width, dims.height);
                  drawMarginGuidesOnCanvas(ctx, dims);

                  if (pageSettings.borderStyle !== 'none') {
                      ctx.strokeStyle = pageSettings.borderColor;
                      ctx.lineWidth = pageSettings.borderStyle === 'thick' ? 5 : 2;
                      ctx.strokeRect(0, 0, dims.width, dims.height);
                  }

                  page.imageBoxes.forEach(ib => {
                      const boxLeft = parseFloat(ib.elem.style.left);
                      const boxTop = parseFloat(ib.elem.style.top);
                      const boxW = ib.elem.offsetWidth;
                      const boxH = ib.elem.offsetHeight;

                      ctx.drawImage(ib.imgElem, boxLeft, boxTop, boxW, boxH);
                      ctx.drawImage(ib.inkCanvas, boxLeft, boxTop, boxW, boxH);
                  });

                  ctx.drawImage(page.canvas, 0, 0, dims.width, dims.height);

                  page.textBoxes.forEach(tb => {
                      const text = tb.contentElem.innerText;
                      if (text.trim().length > 0) {
                          const fontSize = ruleH * 0.72;
                          ctx.font = `${fontSize}px Segoe UI, sans-serif`;
                          
                          if (useGradient) {
                              const grad = ctx.createLinearGradient(0, 0, dims.width, dims.height);
                              grad.addColorStop(0, gradientPrimary);
                              grad.addColorStop(1, gradientSecondary);
                              ctx.fillStyle = grad;
                          } else {
                              ctx.fillStyle = tb.contentElem.style.color || '#000000';
                          }

                          ctx.textBaseline = 'alphabetic';

                          const boxLeft = parseFloat(tb.elem.style.left) || 0;
                          const boxTop = parseFloat(tb.elem.style.top) || 0;
                          const boxWidth = (parseFloat(tb.elem.style.width) || tb.elem.offsetWidth || 220) - 8;

                          const paragraphs = text.split('\n');
                          let currentLineIndex = 0;

                          paragraphs.forEach(paragraph => {
                              const words = paragraph.split(' ');
                              let currentLine = '';

                              words.forEach((word, index) => {
                                  const testLine = currentLine + (currentLine ? ' ' : '') + word;
                                  const metrics = ctx.measureText(testLine);
                                  
                                  if (metrics.width > boxWidth && index > 0) {
                                      let baselineY = boxTop + (currentLineIndex * ruleH) + (ruleH * 0.78);
                                      ctx.fillText(currentLine, boxLeft + 4, baselineY);
                                      currentLine = word;
                                      currentLineIndex++;
                                  } else {
                                      currentLine = testLine;
                                  }
                              });

                              if (currentLine) {
                                  let baselineY = boxTop + (currentLineIndex * ruleH) + (ruleH * 0.78);
                                  ctx.fillText(currentLine, boxLeft + 4, baselineY);
                                  currentLineIndex++;
                              }
                          });
                      }
                  });

                  // Draw NoteKeep Page Branding at Bottom Right
                  const brandX = dims.width - 150;
                  const brandY = dims.height - 18;

                  ctx.save();
                  ctx.font = 'bold 12px Segoe UI, sans-serif';
                  const grad = ctx.createLinearGradient(brandX, brandY, brandX + 70, brandY);
                  grad.addColorStop(0, '#2563eb');
                  grad.addColorStop(1, '#ec4899');
                  ctx.fillStyle = grad;
                  ctx.fillText("📝 NoteKeep", brandX, brandY);

                  ctx.font = 'bold 10px Segoe UI, sans-serif';
                  ctx.fillStyle = '#2563eb';
                  ctx.fillText("Get App", brandX + 80, brandY);
                  ctx.restore();

                  if (pageSettings.showPageNumbers !== false) {
                      ctx.save();
                      ctx.font = 'bold 12px Segoe UI, sans-serif'; ctx.fillStyle = '#6b7280'; ctx.textAlign = 'right';
                      ctx.fillText(`PAGE ${pageIdx + 1}`, dims.width - 20, 25);
                      ctx.restore();
                  }

                  return expCanvas;
              }

      const openAppBtns = document.querySelectorAll('.open-app-btn');
      const notekeepAppOverlay = document.getElementById('notekeepAppOverlay');

      function openApp() {
        notekeepAppOverlay.classList.add('active');
        document.body.classList.add('app-active');
        
        // Push state into history so the user can close the app using device/browser back navigation
        history.pushState({ appOpen: true }, '');
      }

      function closeApp() {
        notekeepAppOverlay.classList.remove('active');
        document.body.classList.remove('app-active');
      }

      openAppBtns.forEach(function(btn) {
        btn.addEventListener('click', function(e) {
          e.preventDefault();
          openApp();
        });
      });

      // Listen for browser/device back navigation gesture or button click
      window.addEventListener('popstate', function() {
        if (notekeepAppOverlay.classList.contains('active')) {
          closeApp();
        }
      });
    });
